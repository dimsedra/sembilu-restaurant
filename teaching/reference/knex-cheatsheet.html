Knex Reference

Setup

npm install knex pg

# Create knexfile.js
npx knex init

Common queries

// SELECT
knex('dishes').where({ branch_id: 1 }).select('*')

// INSERT
knex('orders').insert({ branch_id: 1, table_number: 5, status: 'pending' })

// UPDATE
knex('order_items').where({ id: 42 }).update({ status: 'cooking' })

// DELETE
knex('orders').where({ id: 7 }).del()

// JOIN
knex('orders')
  .join('branches', 'orders.branch_id', 'branches.id')
  .where('branches.city', 'Tegal')
  .select('orders.*')

// COUNT
knex('orders').where({ status: 'pending' }).count('id as total')

// ORDER + LIMIT
knex('orders')
  .where({ branch_id: 1 })
  .orderBy('created_at', 'desc')
  .limit(10)

Column types

table.increments('id')       // auto-incrementing PK
table.string('name', 100)    // VARCHAR(100)
table.text('description')    // TEXT
table.integer('price')       // INTEGER
table.boolean('open')         // BOOLEAN
table.timestamps()            // created_at + updated_at
table.enum('status', ['pending', 'cooking', 'done'])
